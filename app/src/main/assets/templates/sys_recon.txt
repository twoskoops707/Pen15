ENTER
STRING $r += "OS: $((Get-CimInstance Win32_OperatingSystem).Caption)"

ENTER
STRING $r += "User: $env:USERNAME"
ENTER
STRING $r += "Computer: $env:COMPUTERNAME"
ENTER
STRING $r += ""
ENTER
STRING $r += "--- Hardware ---"
ENTER

ENTER

ENTER
STRING $r += (Get-NetIPAddress -AddressFamily IPv4 | Where-Object {$_.InterfaceAlias -notlike "*Loopback*"} | ForEach-Object { "$($_.InterfaceAlias): $($_.IPAddress)" }) -join "`n"

ENTER
STRING $r += "--- Software ---"
ENTER

ENTER
STRING $r += "--- Security ---"
